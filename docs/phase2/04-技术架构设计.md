## 3. 架构概述

### 3.1 架构原则

1. **分层清晰**：对话层、业务层、偏好层职责明确
2. **松耦合**：模块之间通过接口交互，降低依赖
3. **可扩展**：支持水平扩展和功能扩展
4. **高性能**：优化数据库查询和 API 响应
5. **安全可靠**：数据加密、权限控制、错误恢复

### 5.2 智能分配模块

#### 算法流程

```
1. 收集输入
   - 任务列表（预估工时、优先级、截止日期）
   - 用户时间偏好（作息、工作时段）
   - 固定日程
   - 已安排任务

2. 预处理
   - 按优先级和截止日期排序任务
   - 计算可用时间段（排除固定日程和已安排任务）
   - 识别连续时间块

3. 分配任务
   - 为每个任务找到最佳时间段
   - 考虑任务连续性（同一目标的任务连续安排）
   - 预留缓冲时间

4. 冲突检测
   - 检测任务之间是否有时间冲突
   - 如有冲突，调整低优先级任务

5. 返回结果
   - 返回每个任务的建议时间段
   - 标注冲突和建议调整
```

### 5.3 冲突检测模块

#### 算法实现

```python
async def detect_conflict(
    task_date: date,
    start_time: time,
    end_time: time,
    exclude_task_id: int = None
) -> List[Conflict]:
    """
    检测时间冲突

    Args:
        task_date: 任务日期
        start_time: 开始时间
        end_time: 结束时间
        exclude_task_id: 排除的任务ID（用于更新任务时检测冲突）

    Returns:
        冲突列表
    """
    # 查询当天的所有任务和日程
    tasks = await Task.query.filter(
        Task.due_date == task_date,
        Task.start_time.isnot(None),
        Task.end_time.isnot(None)
    ).all()

    events = await Event.query.filter(
        Event.start_date <= task_date,
        Event.end_date >= task_date
    ).all()

    conflicts = []

    # 检测任务冲突
    for task in tasks:
        if task.id == exclude_task_id:
            continue

        if time_overlap(start_time, end_time, task.start_time, task.end_time):
            conflicts.append(Conflict(
                type="task",
                target_id=task.id,
                title=task.title,
                start_time=task.start_time,
                end_time=task.end_time,
                overlap_minutes=calculate_overlap(start_time, end_time, task.start_time, task.end_time)
            ))

    # 检测日程冲突
    for event in events:
        if time_overlap(start_time, end_time, event.start_time, event.end_time):
            conflicts.append(Conflict(
                type="event",
                target_id=event.id,
                title=event.title,
                start_time=event.start_time,
                end_time=event.end_time,
                overlap_minutes=calculate_overlap(start_time, end_time, event.start_time, event.end_time)
            ))

    return conflicts


def time_overlap(start1: time, end1: time, start2: time, end2: time) -> bool:
    """
    检测两个时间段是否有重叠
    """
    return not (end1 <= start2 or start1 >= end2)


def calculate_overlap(start1: time, end1: time, start2: time, end2: time) -> int:
    """
    计算重叠时间（分钟）
    """
    start = max(start1, start2)
    end = min(end1, end2)
    overlap = (end.hour - start.hour) * 60 + (end.minute - start.minute)
    return max(0, overlap)
```

### 5.4 周期性任务模块

#### RRULE 解析

```python
from dateutil.rrule import rrule, rruleset, rrulestr
from dateutil.parser import parse
from datetime import datetime, time, date


def parse_recurrence_rule(recurrence_rule: dict) -> List[date]:
    """
    解析周期性规则，生成日期列表

    Args:
        recurrence_rule: 包含 rrule, exdates, rdates 的字典

    Returns:
        日期列表
    """
    # 创建规则集
    rules = rruleset()

    # 添加重复规则
    if "rrule" in recurrence_rule:
        rules.rrule(rrulestr(recurrence_rule["rrule"]))

    # 添加额外日期
    if "rdates" in recurrence_rule:
        for rdate in recurrence_rule["rdates"]:
            rules.rdate(parse(rdate).date())

    # 排除日期
    if "exdates" in recurrence_rule:
        for exdate in recurrence_rule["exdates"]:
            rules.exdate(parse(exdate).date())

    # 生成所有日期
    dates = list(rules)
    return dates


# 示例
recurrence_rule = {
    "rrule": "FREQ=WEEKLY;BYDAY=MO,WE,FR;INTERVAL=1;UNTIL=20261231T235959Z",
    "exdates": ["2026-03-10", "2026-03-17"],
    "rdates": ["2026-03-08"]
}

dates = parse_recurrence_rule(recurrence_rule)
```

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer

security = HTTPBearer()


async def verify_token(token: str = Depends(security)) -> User:
    """验证 JWT Token"""
    try:
        payload = decode_jwt(token.credentials)
        user_id = payload.get("sub")
        user = await User.query.get(user_id)
        if not user:
            raise HTTPException(status_code=401, detail="User not found")
        return user
    except Exception:
        raise HTTPException(status_code=401, detail="Invalid token")

### 6.2 认证授权

#### JWT Token

```python
import jwt
from datetime import datetime, timedelta

def create_access_token(user_id: int) -> str:
    """创建访问令牌"""
    payload = {
        "sub": str(user_id),
        "exp": datetime.utcnow() + timedelta(days=7),
        "iat": datetime.utcnow()
    }
    token = jwt.encode(payload, settings.SECRET_KEY, algorithm="HS256")
    return token


def decode_jwt(token: str) -> dict:
    """解码 JWT Token"""
    payload = jwt.decode(token, settings.SECRET_KEY, algorithms=["HS256"])
    return payload
```

### 6.3 限流熔断

#### Redis 限流

```python
from redis import Redis
import time

redis = Redis(host=settings.REDIS_HOST, port=6379, db=0)


async def rate_limit(user_id: int, limit: int = 100, window: int = 60) -> bool:
    """
    限流检查

    Args:
        user_id: 用户ID
        limit: 限制次数
        window: 时间窗口（秒）

    Returns:
        是否允许请求
    """
    key = f"rate_limit:{user_id}"
    current = time.time()

    # 移除过期的记录
    redis.zremrangebyscore(key, 0, current - window)

    # 检查当前请求数
    count = redis.zcard(key)

    if count >= limit:
        return False

    # 添加当前请求
    redis.zadd(key, {str(current): current})
    redis.expire(key, window)

    return True
```

### 8.2 缓存策略

#### Redis 缓存

```python
import json
from redis import Redis

redis = Redis(host=settings.REDIS_HOST, port=6379, db=0)


async def cache_get(key: str) -> Optional[dict]:
    """从缓存获取数据"""
    data = redis.get(key)
    if data:
        return json.loads(data)
    return None


async def cache_set(key: str, value: dict, ttl: int = 3600):
    """设置缓存"""
    redis.setex(key, ttl, json.dumps(value))


async def get_conversation_cached(conversation_id: int) -> Optional[Conversation]:
    """获取对话会话（带缓存）"""
    key = f"conversation:{conversation_id}"
    cached = await cache_get(key)
    if cached:
        return cached

    conversation = await Conversation.query.get(conversation_id)
    if conversation:
        await cache_set(key, conversation.to_dict(), ttl=300)

    return conversation
```
