- 显示 AI 回复（支持流式）
- 显示操作建议卡片

**Props**：
```javascript

  conversationId: String,    // 对话ID
  conversationType: String, // 对话类型
  disabled: Boolean          // 是否禁用
}
```

**Events**：
```javascript
{
  'operation-confirm': (operation) => {},  // 确认操作

}
```

**使用示例**：
```vue
<ChatPanel
  :conversation-id="conversationId"
  :conversation-type="'goal_planning'"
```

## 5. 状态管理

### 5.1 User Store

```javascript
// store/user.js
import { defineStore } from 'pinia'

export const useUserStore = defineStore('user', {
  state: () => ({
    user: null,
    token: null,
    preferences: null
  }),

  actions: {
    setUser(user) {
      this.user = user
    },

    setToken(token) {
      this.token = token
      localStorage.setItem('token', token)
    },

    clear() {
      this.user = null
      this.token = null
      localStorage.removeItem('token')
    }
  }
})
```

### 5.2 Conversation Store

```javascript
// store/conversation.js
import { defineStore } from 'pinia'
import { createConversation, sendMessage } from '@/api/conversation'

export const useConversationStore = defineStore('conversation', {
  state: () => ({
    conversations: [],
    currentConversation: null,
    messages: [],
    isTyping: false
  }),

  actions: {
    async startConversation(type) {
      const { data } = await createConversation({ conversation_type: type })
      this.currentConversation = data
      this.messages = []
      return data
    },

    async sendMessage(content) {
      // 添加用户消息
      this.messages.push({
        role: 'user',
        content,
        created_at: new Date()
      })

      // 设置正在输入
      this.isTyping = true

      // 发送请求
      const { data } = await sendMessage(
        this.currentConversation.id,
        { content }
      )

      // 添加 AI 回复
      this.messages.push({
        role: 'assistant',
        content: data.content,
        content_json: data.content_json,
        created_at: new Date()
      })

      this.isTyping = false
    }
  }
})
```

## 6. API 接口封装

### 6.1 请求封装

```javascript
// utils/request.js
import axios from 'axios'
import { useUserStore } from '@/store/user'

const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000
})

// 请求拦截器
request.interceptors.request.use(
  (config) => {
    const userStore = useUserStore()
    if (userStore.token) {
      config.headers.Authorization = `Bearer ${userStore.token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
request.interceptors.response.use(
  (response) => {
    return response.data
  },
  (error) => {
    if (error.response?.status === 401) {
      // Token 过期，跳转到登录页
      const userStore = useUserStore()
      userStore.clear()
      window.location.href = '/login'
    }
    return Promise.reject(error)
  }
)

export default request
```

## 7. 路由配置

```javascript
// router/index.js
import { createRouter, createWebHistory } from 'vue-router'
import { useUserStore } from '@/store/user'

const routes = [
  {
    path: '/',
    component: () => import('@/views/Home.vue')
  },
  {
    path: '/conversation',
    component: () => import('@/views/Conversation.vue'),
    meta: { requiresAuth: true }
  },
  {
    path: '/schedule',
    component: () => import('@/views/Schedule.vue'),
    meta: { requiresAuth: true }
  },
  {
    path: '/tasks',
    component: () => import('@/views/TaskList.vue'),
    meta: { requiresAuth: true }
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

// 路由守卫
router.beforeEach((to, from, next) => {
  const userStore = useUserStore()
  
  if (to.meta.requiresAuth && !userStore.token) {
    next('/login')
  } else {
    next()
  }
})

export default router
```

## 12. 常见问题

### Q1: 如何处理流式响应？

使用 EventSource 或 Fetch API 的流式响应：

```javascript
async function sendMessageStream(conversationId, content) {
  const response = await fetch(
    `/api/conversations/${conversationId}/messages/stream`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ content })
    }
  )

  const reader = response.body.getReader()
  const decoder = new TextDecoder()

  while (true) {
    const { done, value } = await reader.read()
    if (done) break

    const text = decoder.decode(value)
    // 处理流式数据
  }
}
```

### Q2: 如何实现拖拽调整时间？

使用 HTML5 拖放 API 或第三方库（如 @draggable/core）：

```vue
<template>
  <div
    class="task-card"
    draggable="true"
    @dragstart="handleDragStart"
    @dragend="handleDragEnd"
  >
    {{ task.title }}
  </div>
</template>

<script>
export default {
  methods: {
    handleDragStart(e) {
      e.dataTransfer.setData('taskId', this.task.id)
    },
    handleDragEnd(e) {
      // 处理拖拽结束
    }
  }
}
</script>
```
